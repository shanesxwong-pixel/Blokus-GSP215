    // Standard 21 Blokus Polyominoes definitions

      [[1]], // 1 Monomino

    function selectPiece(idx) {
      selectedPieceIdx = idx;
      currentPieceMatrix = SHAPES[idx];
      renderPieceRack();
    }

    function rotateSelectedPiece() {
      if (!currentPieceMatrix) return;
      const rows = currentPieceMatrix.length;
      const cols = currentPieceMatrix[0].length;
      let rotated = Array(cols).fill(0).map(() => Array(rows).fill(0));
      for (let r = 0; r < rows; r++) {
        for (let c = 0; c < cols; c++) {
          rotated[c][rows - 1 - r] = currentPieceMatrix[r][c];
        }
      }
      currentPieceMatrix = rotated;
    }

    function flipSelectedPiece() {
      if (!currentPieceMatrix) return;
      currentPieceMatrix = currentPieceMatrix.map(row => [...row].reverse());
    }

    // Move Validation Logic
    function isValidMove(b, pId, matrix, startR, startC) {
      const rows = matrix.length;
      const cols = matrix[0].length;
      let touchesCorner = false;

      // Starting corners for P1, P2, P3, P4
      const corners = [[0,0], [0,19], [19,19], [19,0]];
      const startCorner = corners[pId - 1];

      for (let r = 0; r < rows; r++) {
        for (let c = 0; c < cols; c++) {
          if (matrix[r][c] === 1) {
            const br = startR + r;
            const bc = startC + c;

            if (br < 0 || br >= 20 || bc < 0 || bc >= 20) return false; // Out of bounds
            if (b[br][bc] !== 0) return false; // Overlap

            // Check edge collisions (Same color cannot touch edges)
            const edges = [[br-1, bc], [br+1, bc], [br, bc-1], [br, bc+1]];
            for (let [er, ec] of edges) {
              if (er >= 0 && er < 20 && ec >= 0 && ec < 20) {
                if (b[er][ec] === pId) return false;
              }
            }

            // Check first move corner or diagonal touch
            if (br === startCorner[0] && bc === startCorner[1]) {
              touchesCorner = true;
            }

            const diagCorners = [[br-1, bc-1], [br-1, bc+1], [br+1, bc-1], [br+1, bc+1]];
            for (let [dr, dc] of diagCorners) {
              if (dr >= 0 && dr < 20 && dc >= 0 && dc < 20) {
                if (b[dr][dc] === pId) touchesCorner = true;
              }
            }
          }
        }
      }
      return touchesCorner;
    }

    function handleCellClick(r, c) {
      const p = players[currentTurn];
      if (p.type === "ai" || selectedPieceIdx === null) return;

      if (isValidMove(board, p.id, currentPieceMatrix, r, c)) {
        placePiece(p.id, currentPieceMatrix, r, c);
        p.pieces = p.pieces.filter(idx => idx !== selectedPieceIdx);
        selectedPieceIdx = null;
        currentPieceMatrix = null;
        advanceTurn();
      }
    }

    function placePiece(pId, matrix, startR, startC) {
      for (let r = 0; r < matrix.length; r++) {
        for (let c = 0; c < matrix[0].length; c++) {
          if (matrix[r][c] === 1) {
            board[startR + r][startC + c] = pId;
            const cell = document.querySelector(`.cell[data-r="${startR + r}"][data-c="${startC + c}"]`);
            if (cell) cell.className = `cell p${pId}`;
          }
        }
      }
    }

    function passTurn() {
      players[currentTurn].passed = true;
      advanceTurn();
    }

    function advanceTurn() {
      if (players.every(p => p.passed || p.pieces.length === 0)) {
        endGame();
        return;
      }

      do {
        currentTurn = (currentTurn + 1) % 4;
      } while (players[currentTurn].passed && !players.every(p => p.passed));

      updateTurnUI();

      if (players[currentTurn].type === "ai" && !players[currentTurn].passed) {
        setTimeout(playAITurn, 400);
      }
    }

    // Basic Strategy AI
    function playAITurn() {
      const p = players[currentTurn];
      let validMoves = [];

      p.pieces.forEach(pieceIdx => {
        let shape = SHAPES[pieceIdx];
        for (let rot = 0; rot < 4; rot++) {
          for (let r = 0; r < 20; r++) {
            for (let c = 0; c < 20; c++) {
              if (isValidMove(board, p.id, shape, r, c)) {
                validMoves.push({ pieceIdx, shape, r, c });
              }
            }
          }
          // Rotate matrix
          shape = rotateMatrix(shape);
        }
      });

      if (validMoves.length > 0) {
        // AI prefers placing larger pieces first
        validMoves.sort((a, b) => countBlocks(b.shape) - countBlocks(a.shape));
        const move = validMoves[0];
        placePiece(p.id, move.shape, move.r, move.c);
        p.pieces = p.pieces.filter(idx => idx !== move.pieceIdx);
      } else {
        p.passed = true;
      }

      advanceTurn();
    }

    function rotateMatrix(mat) {
      const rows = mat.length, cols = mat[0].length;
      let res = Array(cols).fill(0).map(() => Array(rows).fill(0));
      for (let r = 0; r < rows; r++) {
        for (let c = 0; c < cols; c++) res[c][rows - 1 - r] = mat[r][c];
      }
      return res;
    }

    function countBlocks(mat) {
      return mat.flat().reduce((a, b) => a + b, 0);
    }

    // Audio Generator (Procedural Lo-Fi Ambient)
    function toggleAudio() {
      const widget = document.getElementById("lofi-widget");
      if (!isAudioPlaying) {
        if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        isAudioPlaying = true;
        widget.style.display = "flex";
        playLoFiBeat();
      } else {
        isAudioPlaying = false;
        widget.style.display = "none";
      }
    }

    function playLoFiBeat() {
      if (!isAudioPlaying) return;
      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();
      osc.type = 'sine';
      osc.frequency.setValueAtTime(220 + Math.random() * 80, audioCtx.currentTime);
      gain.gain.setValueAtTime(0.05, audioCtx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 1.5);
      osc.connect(gain);
      gain.connect(audioCtx.destination);
      osc.start();
      osc.stop(audioCtx.currentTime + 1.5);
      setTimeout(playLoFiBeat, 2000);
    }

    // Game End & Debrief Logic
    async function endGame() {
      players.forEach(p => {
        const remainingBlocks = p.pieces.reduce((sum, idx) => sum + countBlocks(SHAPES[idx]), 0);
        p.score = -remainingBlocks; // Blokus scoring formula
      });

      players.sort((a, b) => b.score - a.score);

      const statsEl = document.getElementById("debrief-stats");
      statsEl.innerHTML = players.map((p, idx) => `
        <div style="display:flex; justify-content:space-between; margin-bottom: 6px;">
          <span>#${idx+1} ${p.name}:</span>
          <strong>${p.score} pts</strong>
        </div>
      `).join("");

      document.getElementById("debrief-modal").classList.add("active");

      // Save to Storage API
      try {
        if (window.Storage) {
          let history = (await Storage.get("blokus_game_history")) || [];
          history.unshift({
            date: new Date().toLocaleDateString(),
            winner: players[0].name,
            scores: players.map(p => `${p.name}: ${p.score}`)
          });
          await Storage.set("blokus_game_history", history.slice(0, 10));
        }
      } catch (err) {
        console.error(err);
      }
    }

    function closeDebrief() {
      document.getElementById("debrief-modal").classList.remove("active");
      navigateTo("screen-setup");
    }

    async function showHistory() {
      navigateTo("screen-history");
      const list = document.getElementById("history-list");
      list.innerHTML = "Loading...";

      try {
        let history = [];
        if (window.Storage) {
          history = (await Storage.get("blokus_game_history")) || [];
        }

        if (history.length === 0) {
          list.innerHTML = `<div style="color: var(--text-muted);">No match history available yet.</div>`;
          return;
        }

        list.innerHTML = history.map(item => `
          <div class="setup-card">
            <div style="font-weight: bold; margin-bottom: 4px;">Winner: ${item.winner} (${item.date})</div>
            <div style="font-size: 0.8rem; color: var(--text-muted);">${item.scores.join(" | ")}</div>
          </div>
        `).join("");
      } catch (e) {
        list.innerHTML = "Failed to load history.";
      }
    }

    function confirmEndGame() {
      if (confirm("End current simulation and debrief?")) endGame();
    }
